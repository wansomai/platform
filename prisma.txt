generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Account {
  id                String  @id @default(cuid())
  userId            String
  type              String
  provider          String
  providerAccountId String
  refresh_token     String?
  access_token      String?
  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String?
  session_state     String?
  user              User    @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerAccountId])
}

model Session {
  id           String   @id @default(cuid())
  sessionToken String   @unique
  userId       String
  expires      DateTime
  user         User     @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model VerificationToken {
  identifier String
  token      String   @unique
  expires    DateTime

  @@unique([identifier, token])
}

model User {
  id                    String                  @id @default(cuid())
  email                 String                  @unique
  fullName              String?
  password              String
  role                  String                  @default("user")
  organizationId        String
  createdAt             DateTime                @default(now())
  updatedAt             DateTime                @updatedAt
  accounts              Account[]
  actions               ConversationAction[]
  conversationDocuments ConversationDocument[]
  documents             Document[]
  Folder                Folder[]
  sentInvitations       Invitation[]            @relation("InvitedBy")
  messages              Message[]
  projectDocuments      ProjectDocument[]
  projectMembers        ProjectMember[]
  sessions              Session[]
  SharedMessage         SharedMessage[]
  SharedWorkspace       SharedWorkspace[]
  SharedWorkspaceAccess SharedWorkspaceAccess[]
  organization          Organization            @relation(fields: [organizationId], references: [id])
  createdAIAssociates   AIAssociate[]
}

model Organization {
  id           String        @id @default(cuid())
  name         String
  createdAt    DateTime      @default(now())
  updatedAt    DateTime      @updatedAt
  documents    Document[]
  Folder       Folder[]
  invitations  Invitation[]
  projects     Project[]
  users        User[]
  aiAssociates AIAssociate[]
}

model Project {
  id               String             @id @default(cuid())
  title            String
  description      String?
  status           String             @default("active")
  organizationId   String
  createdAt        DateTime           @default(now())
  updatedAt        DateTime           @updatedAt
  conversations    Conversation[]
  documents        Document[]
  events           Event[]
  invitations      Invitation[]
  knowledgeBase    KnowledgeBase?
  organization     Organization       @relation(fields: [organizationId], references: [id])
  projectDocuments ProjectDocument[]
  members          ProjectMember[]
  SharedWorkspace  SharedWorkspace[]
  associates       ProjectAssociate[]
}

model ProjectMember {
  userId    String
  projectId String
  role      String   @default("member")
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  project   Project  @relation(fields: [projectId], references: [id])
  user      User     @relation(fields: [userId], references: [id])

  @@id([userId, projectId])
}

model Document {
  id                     String                   @id @default(cuid())
  project_id             String?                  @map("projectId")
  organization_id        String                   @map("organizationId")
  title                  String
  description            Json?
  file_url               String                   @map("fileUrl")
  file_type              String                   @map("fileType")
  file_size              Int                      @map("fileSize")
  status                 String                   @default("active")
  section                String                   @default("general")
  created_by             String                   @map("createdBy")
  metadata               Json?
  content_extracted      Json?                    @map("contentExtracted")
  created_at             DateTime                 @default(now()) @map("createdAt")
  updated_at             DateTime                 @updatedAt @map("updatedAt")
  folderId               String?
  conversationReferences ConversationDocument[]
  createdByUser          User                     @relation(fields: [created_by], references: [id])
  folder                 Folder?                  @relation(fields: [folderId], references: [id])
  organization           Organization             @relation(fields: [organization_id], references: [id])
  project                Project?                 @relation(fields: [project_id], references: [id])
  content                DocumentContent?
  embeddings             Embedding[]
  messageReferences      MessageReference[]
  projectReferences      ProjectDocument[]
  SharedMessageReference SharedMessageReference[]

  @@index([organization_id])
}

model Folder {
  id             String       @id @default(cuid())
  name           String
  organizationId String
  parentId       String?
  createdBy      String
  createdAt      DateTime     @default(now())
  updatedAt      DateTime     @updatedAt
  documents      Document[]
  user           User         @relation(fields: [createdBy], references: [id])
  organization   Organization @relation(fields: [organizationId], references: [id])
  parent         Folder?      @relation("FolderHierarchy", fields: [parentId], references: [id])
  children       Folder[]     @relation("FolderHierarchy")
}

model ProjectDocument {
  project_id  String   @map("projectId")
  document_id String   @map("documentId")
  added_at    DateTime @default(now()) @map("addedAt")
  added_by    String   @map("addedBy")
  user        User     @relation(fields: [added_by], references: [id])
  document    Document @relation(fields: [document_id], references: [id], onDelete: Cascade)
  project     Project  @relation(fields: [project_id], references: [id], onDelete: Cascade)

  @@id([project_id, document_id])
}

model ConversationDocument {
  conversation_id String       @map("conversationId")
  document_id     String       @map("documentId")
  added_at        DateTime     @default(now()) @map("addedAt")
  added_by        String       @map("addedBy")
  user            User         @relation(fields: [added_by], references: [id])
  conversation    Conversation @relation(fields: [conversation_id], references: [id], onDelete: Cascade)
  document        Document     @relation(fields: [document_id], references: [id], onDelete: Cascade)

  @@id([conversation_id, document_id])
}

model DocumentContent {
  documentId String   @id
  content    String
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt
  document   Document @relation(fields: [documentId], references: [id], onDelete: Cascade)
}

model Embedding {
  id         String   @id @default(cuid())
  documentId String
  vector     String
  chunkText  String
  chunkIndex Int
  createdAt  DateTime @default(now())
  document   Document @relation(fields: [documentId], references: [id], onDelete: Cascade)
}

model Conversation {
  id                 String                 @id @default(cuid())
  title              String
  projectId          String
  isPinned           Boolean                @default(false)
  createdAt          DateTime               @default(now())
  updatedAt          DateTime               @updatedAt
  project            Project                @relation(fields: [projectId], references: [id], onDelete: Cascade)
  actions            ConversationAction[]
  documentReferences ConversationDocument[]
  meta               ConversationMeta?
  messages           Message[]
  SharedWorkspace    SharedWorkspace[]
  aiAssociate        AIAssociate?           @relation(fields: [aiAssociateId], references: [id])
  aiAssociateId      String?
}

model Message {
  id             String             @id @default(cuid())
  conversationId String
  content        String
  role           String             @default("user")
  createdAt      DateTime           @default(now())
  userId         String?
  metadata       Json?
  conversation   Conversation       @relation(fields: [conversationId], references: [id], onDelete: Cascade)
  user           User?              @relation(fields: [userId], references: [id])
  references     MessageReference[]
}

model MessageReference {
  id         String   @id @default(cuid())
  messageId  String
  documentId String
  text       String
  page       Int?
  createdAt  DateTime @default(now())
  document   Document @relation(fields: [documentId], references: [id], onDelete: Cascade)
  message    Message  @relation(fields: [messageId], references: [id], onDelete: Cascade)
}

model Event {

  createdAt   DateTime @default(now())

model KnowledgeBase {

  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  projectId      String

model ConversationMeta {

  settings       Json?
  createdAt      DateTime     @default(now())
  updatedAt      DateTime     @updatedAt
  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)

  parameters     String
  status         String       @default("pending")
  resultContent  String?
  resultUrl      String?
  createdAt      DateTime     @default(now())
  updatedAt      DateTime     @updatedAt
  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)

model SharedMessage {
  id                     String                   @id
  sharedWorkspaceId      String
  sessionId              String
  content                String
  role                   String                   @default("user")
  userName               String
  userId                 String?
  createdAt              DateTime                 @default(now())
  SharedWorkspace        SharedWorkspace          @relation(fields: [sharedWorkspaceId], references: [id], onDelete: Cascade)

}

model SharedMessageReference {
  id            String        @id
  messageId     String
  documentId    String
  text          String
  page          Int?
  createdAt     DateTime      @default(now())
  Document      Document      @relation(fields: [documentId], references: [id], onDelete: Cascade)

model SharedWorkspace {
  id                    String                  @id
  projectId             String
  conversationId        String
  createdBy             String
  title                 String
  description           String?
  accessMode            String                  @default("read_only")
  isActive              Boolean                 @default(true)
  expiresAt             DateTime?
  passcode              String?
  createdAt             DateTime                @default(now())
  updatedAt             DateTime
  SharedMessage         SharedMessage[]
  Conversation          Conversation            @relation(fields: [conversationId], references: [id], onDelete: Cascade)

  SharedWorkspaceAccess SharedWorkspaceAccess[]
}

model SharedWorkspaceAccess {
  id                String          @id
  sharedWorkspaceId String
  userId            String?
  userEmail         String?
  userName          String
  accessTime        DateTime        @default(now())
  SharedWorkspace   SharedWorkspace @relation(fields: [sharedWorkspaceId], references: [id], onDelete: Cascade)
  User              User?           @relation(fields: [userId], references: [id])
}

model AIAssociate {
  id             String   @id @default(cuid())
  name           String
  instructions   String   @db.Text
  organizationId String
  createdById    String
  createdAt      DateTime @default(now())
  updatedAt      DateTime @updatedAt

  // Relations
  organization  Organization       @relation(fields: [organizationId], references: [id])
  createdBy     User               @relation(fields: [createdById], references: [id])
  steps         AssociateStep[]
  tools         AssociateTool[]
  projects      ProjectAssociate[] // Many-to-many relation
  conversations Conversation[]
}

model AssociateStep {
  id          String @id @default(cuid())
  associateId String
  description String
  stepOrder   Int // To maintain the order of steps

  // Relations
  associate AIAssociate @relation(fields: [associateId], references: [id], onDelete: Cascade)
}

model AssociateTool {
  id          String @id @default(cuid())
  associateId String
  toolId      String // The ID of the tool (e.g., "documentSearch", "webSearch")

  // Relations
  associate AIAssociate @relation(fields: [associateId], references: [id], onDelete: Cascade)
}

model ProjectAssociate {
  associateId String
  projectId   String
  addedAt     DateTime @default(now())

  // Relations
  associate AIAssociate @relation(fields: [associateId], references: [id], onDelete: Cascade)
  project   Project     @relation(fields: [projectId], references: [id], onDelete: Cascade)

  @@id([associateId, projectId])
}
